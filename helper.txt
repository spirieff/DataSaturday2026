Average daily revenue per store

SELECT store_name, ROUND(AVG(daily_rev), 2) AS avg_daily_revenue 
FROM (SELECT store_name, transaction_date, SUM(line_item_amount) AS daily_rev 
      FROM dev_overwatch.coffee_gold.v_coffee_performance GROUP BY store_name, transaction_date) 
GROUP BY store_name ORDER BY avg_daily_revenue DESC



Beverage percentage by store:
SELECT store_name, 
  ROUND(SUM(CASE WHEN product_group = 'Beverages' THEN line_item_amount ELSE 0 END) / SUM(line_item_amount) * 100, 1) AS beverage_pct 
FROM dev_overwatch.coffee_gold.v_coffee_performance GROUP BY store_name ORDER BY beverage_pct DESC
